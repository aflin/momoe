# momoe-metal: Momoe's engine for Apple Silicon, built with Momoe's copy of ggml (patched: patch_ggml_pause.py).
# On Linux the same engine builds against ggml's CUDA backend as momoe-cuda (MOMOE_CUDA): no pause points there, the
# scheduler's callback stops after each layer's router instead.
cmake_minimum_required(VERSION 3.21)
if(APPLE)
  project(momoe_metal C CXX OBJC)
  set(GGML_METAL ON CACHE BOOL "" FORCE)
  set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
else()
  project(momoe_cuda C CXX)
  set(GGML_CUDA ON CACHE BOOL "" FORCE)
  set(GGML_CUDA_GRAPHS ON CACHE BOOL "" FORCE)           # a reused decode graph is replayed (ggml: off unless asked)
endif()
set(CMAKE_CXX_STANDARD 17)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(ggml)
if(APPLE)
  add_executable(momoe-metal momoe_metal.cpp)
  target_link_libraries(momoe-metal PRIVATE ggml)
else()
  add_executable(momoe-cuda momoe_metal.cpp)
  find_package(CUDAToolkit REQUIRED)
  target_compile_definitions(momoe-cuda PRIVATE MOMOE_CUDA=1)
  target_link_libraries(momoe-cuda PRIVATE ggml CUDA::cudart)
endif()
